#include <OneWire.h>
#include <DallasTemperature.h>
#include <Adafruit_ADS1X15.h>
#include <Wire.h>
#include <ESP8266WiFi.h>
#include <ESP8266WebServer.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>

// Sensor Pins
#define ONE_WIRE_BUS 0
#define TdsSensorPin A0
#define UVC_LED_PIN D5        // GPIO 14
#define HEATER_PIN D6         // GPIO 12

// Settings
#define VREF 3.3
#define SCOUNT 30
#define SCREEN_WIDTH 128
#define SCREEN_HEIGHT 64
#define OLED_RESET -1
#define SCREEN_ADDRESS 0x3C

// WiFi AP credentials
const char* ssid = "ESP8266_PureStream";
const char* password = "";

// Initialize objects
OneWire oneWire(ONE_WIRE_BUS);
DallasTemperature sensors(&oneWire);
Adafruit_ADS1115 ads;
Adafruit_SSD1306 display(SCREEN_WIDTH, SCREEN_HEIGHT, &Wire, OLED_RESET);
ESP8266WebServer server(80);

// Sensor variables
int analogBuffer[SCOUNT];
int analogBufferTemp[SCOUNT];
int analogBufferIndex = 0;
int copyIndex = 0;

float averageVoltage = 0;
float tdsValue = 0;
float turbidityVoltage = 0;
float temperature = 0;
float turbidityNTU = 0;

// State variables
bool uvcState = false;
bool heaterState = false;

// ===== HELPER FUNCTIONS =====

int getMedianNum(int bArray[], int iFilterLen) {
  int bTab[iFilterLen];
  for (byte i = 0; i < iFilterLen; i++) bTab[i] = bArray[i];
  int i, j, bTemp;
  for (j = 0; j < iFilterLen - 1; j++) {
    for (i = 0; i < iFilterLen - j - 1; i++) {
      if (bTab[i] > bTab[i + 1]) {
        bTemp = bTab[i];
        bTab[i] = bTab[i + 1];
        bTab[i + 1] = bTemp;
      }
    }
  }
  if ((iFilterLen & 1) > 0)
    bTemp = bTab[(iFilterLen - 1) / 2];
  else
    bTemp = (bTab[iFilterLen / 2] + bTab[iFilterLen / 2 - 1]) / 2;
  return bTemp;
}

void displayOLED() {
  display.clearDisplay();
  
  // Title
  display.setTextSize(1);
  display.setTextColor(SSD1306_WHITE);
  display.setCursor(0, 0);
  display.println("PureStream Monitor");
  display.drawLine(0, 10, 128, 10, SSD1306_WHITE);
  
  // Temperature
  display.setCursor(0, 16);
  if (temperature == -127.0) {
    display.println("Temp: ERROR!");
  } else {
    display.print("Temp: ");
    display.print(temperature, 1);
    display.println("C");
  }
  
  // TDS
  display.print("TDS: ");
  display.print((int)tdsValue);
  display.println(" ppm");
  
  // Turbidity
  display.print("Turb: ");
  display.print(turbidityNTU);
  display.println(" NTU");
  
  // Status Line
  display.setCursor(0, 55);
  display.print("H:");
  display.print(heaterState ? "ON " : "OFF");
  display.print(" U:");
  display.println(uvcState ? "ON" : "OFF");
  
  display.display();
}

void updateSensorReadings() {
  // Read TDS analog value every 40ms
  static unsigned long analogSampleTimepoint = millis();
  if (millis() - analogSampleTimepoint > 40U) {
    analogSampleTimepoint = millis();
    analogBuffer[analogBufferIndex] = analogRead(TdsSensorPin);
    analogBufferIndex++;
    if (analogBufferIndex == SCOUNT) analogBufferIndex = 0;
  }

  // Calculate TDS and read blocking sensors every 800ms
  static unsigned long printTimepoint = millis();
  if (millis() - printTimepoint > 800U) {
    printTimepoint = millis();

    // Read Temperature (Moved inside the 800ms timer)
    sensors.requestTemperatures();
    temperature = sensors.getTempCByIndex(0);

    for (copyIndex = 0; copyIndex < SCOUNT; copyIndex++)
      analogBufferTemp[copyIndex] = analogBuffer[copyIndex];

    averageVoltage = getMedianNum(analogBufferTemp, SCOUNT) * VREF / 1024.0;

    float compensationCoefficient = 1.0 + 0.02 * (temperature - 25.0);
    float compensationVoltage = averageVoltage / compensationCoefficient;

    tdsValue = (133.42 * compensationVoltage * compensationVoltage * compensationVoltage
                - 255.86 * compensationVoltage * compensationVoltage
                + 857.39 * compensationVoltage) * 0.5;

    // Read Turbidity from ADS1115
    int16_t adc0 = ads.readADC_SingleEnded(0);
    turbidityVoltage = adc0 * 0.1875 / 1000.0;
    turbidityNTU = (turbidityVoltage < 2.5) ? (2.5 - turbidityVoltage) * 10 : 0;

    if (temperature == -127.0) {
      Serial.println("Temperature sensor not detected!");
    } else {
      Serial.print("Temp: ");
      Serial.print(temperature, 1);
      Serial.print("C | TDS: ");
      Serial.print((int)tdsValue);
      Serial.print(" ppm | Turb: ");
      Serial.print(turbidityNTU);
      Serial.println(" NTU");
    }
    
    displayOLED();
  }
}

// ===== WEB SERVER HANDLERS =====

void handleGetData() {
  String json = "{";
  json += "\"temperature\":" + String(temperature, 1) + ",";
  json += "\"tds\":" + String((int)tdsValue) + ",";
  json += "\"turbidity\":" + String(turbidityNTU) + ",";
  json += "\"heaterState\":" + String(heaterState ? "true" : "false") + ",";
  json += "\"uvcState\":" + String(uvcState ? "true" : "false");
  json += "}";
  
  server.sendHeader("Access-Control-Allow-Origin", "*");
  server.send(200, "application/json", json);
}

void handleUVC() {
  if (server.hasArg("state")) {
    String state = server.arg("state");
    
    if (state == "on") {
      digitalWrite(UVC_LED_PIN, HIGH);
      uvcState = true;
      Serial.println("UVC: ON");
    } else if (state == "off") {
      digitalWrite(UVC_LED_PIN, LOW);
      uvcState = false;
      Serial.println("UVC: OFF");
    }
    
    String response = "{\"uvcState\":" + String(uvcState ? "true" : "false") + "}";
    server.sendHeader("Access-Control-Allow-Origin", "*");
    server.send(200, "application/json", response);
  } else {
    server.send(400, "text/plain", "Missing state parameter");
  }
}

void handleHeater() {
  if (server.hasArg("state")) {
    String state = server.arg("state");
    
    if (state == "on") {
      digitalWrite(HEATER_PIN, HIGH);
      heaterState = true;
      Serial.println("Heater: ON");
    } else if (state == "off") {
      digitalWrite(HEATER_PIN, LOW);
      heaterState = false;
      Serial.println("Heater: OFF");
    }
    
    String response = "{\"heaterState\":" + String(heaterState ? "true" : "false") + "}";
    server.sendHeader("Access-Control-Allow-Origin", "*");
    server.send(200, "application/json", response);
  } else {
    server.send(400, "text/plain", "Missing state parameter");
  }
}

// ===== SETUP =====

void setup() {
  Serial.begin(9600);
  delay(1000);
  
  // Initialize sensors
  sensors.begin();
  
  // Initialize pins
  pinMode(TdsSensorPin, INPUT);
  pinMode(UVC_LED_PIN, OUTPUT);
  pinMode(HEATER_PIN, OUTPUT);
  digitalWrite(UVC_LED_PIN, LOW);
  digitalWrite(HEATER_PIN, LOW);

  // Initialize OLED
  if (!display.begin(SSD1306_SWITCHCAPVCC, SCREEN_ADDRESS)) {
    Serial.println("SSD1306 allocation failed");
    while (1);
  }
  
  display.clearDisplay();
  display.setTextSize(1);
  display.setTextColor(SSD1306_WHITE);
  display.setCursor(0, 20);
  display.println("Initializing...");
  display.display();

  // Initialize ADS1115
  if (!ads.begin()) {
    Serial.println("Failed to initialize ADS1115. Check wiring!");
    display.clearDisplay();
    display.setCursor(0, 20);
    display.println("ADS1115 Init Failed");
    display.display();
    while (1);
  }
  Serial.println("ADS1115 connected successfully.");

  // Start WiFi AP
  WiFi.mode(WIFI_AP);
  WiFi.softAP(ssid, password);
  delay(1000);
  
  Serial.println("\n\nWiFi AP started!");
  Serial.print("SSID: ");
  Serial.println(ssid);
  Serial.print("IP: ");
  Serial.println(WiFi.softAPIP());

  // Setup HTTP routes
  server.on("/data", HTTP_GET, handleGetData);
  server.on("/uvc", handleUVC);
  server.on("/heater", handleHeater);
  server.begin();
  
  Serial.println("Web server started!");
  delay(1000);
  displayOLED();
}

// ===== MAIN LOOP =====

void loop() {
  server.handleClient();
  updateSensorReadings();
}
